project(latency_breakdown LANGUAGES C CXX)

set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
get_filename_component(WORK "${CMAKE_BINARY_DIR}/.." ABSOLUTE)
set(LIBIPC_PROJECT_DIR "${WORK}/source")
set(LIBIPC_BUILD_SHARED_LIBS ON)
set(ROSBUILD OFF)
set(ENABLE_DEBUG_INFO OFF)
include(GNUInstallDirs)
add_subdirectory("${LIBIPC_PROJECT_DIR}/src" "${CMAKE_BINARY_DIR}/ipc")
set(DDS_ROOT "/home/zwc/branch/lejulab_platform/src/lejusdk/3rd_party/x86_64/cyclonedds-0.10.2" CACHE PATH "DDS 安装")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../transport_comparison/generate_types.py"
  "${CMAKE_BINARY_DIR}" "${DDS_ROOT}/bin/idlc" COMMAND_ERROR_IS_FATAL ANY)
add_executable(comparison "${WORK}/comparison.cpp" "${CMAKE_BINARY_DIR}/comparison_types.c")

target_link_libraries(comparison PRIVATE ipc "${DDS_ROOT}/lib/libddsc.so" pthread rt)
target_compile_options(comparison PRIVATE -O3 -DNDEBUG)
set_target_properties(comparison PROPERTIES BUILD_RPATH "${DDS_ROOT}/lib;${CMAKE_BINARY_DIR}/lib")
