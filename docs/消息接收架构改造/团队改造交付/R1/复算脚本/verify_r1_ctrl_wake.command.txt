# 4) state3 的 idle_exits / thread_restarts（r25 无该读数，独立补齐；字段不受诊断门控）
g++ -std=c++17 -O2 -DNDEBUG -I include -I src -I 3rdparty \
    -o build/t32/state3idle docs/消息接收架构改造/团队改造交付/R1/复算脚本/verify_r1_state3_idle.cpp \
    -L build/lib -lipc -lpthread -Wl,-rpath,$PWD/build/lib
./build/t32/state3idle 1000 60
#   实测：idle_exits_delta=0 thread_restarts_delta=0 wait_timeouts=19193 sched wakes/s=8807.1

# 5) 决定性分解：把稳态 CPU 的两个候选成本源**按原生计数**换算成同一量纲（项访问/s）
#    ⛔ 不依赖「关池」反事实（关池会换成 1000 条兼容收包线程，本身就是混淆）
g++ -std=c++17 -O2 -DNDEBUG -I include -I src -I 3rdparty \
    -o build/t32/decomp docs/消息接收架构改造/团队改造交付/R1/复算脚本/verify_r1_work_decomposition.cpp \
    -L build/lib -lipc -lpthread -Wl,-rpath,$PWD/build/lib
./build/t32/decomp 1000 60
#   实测：cpu=0.5958 core
#         控制面 ticks=8564.5/s × entries=2000 × 2 全扫 = 3.426e7 项访问/s
#         接收池 scan_rounds=352/s × scanned_routes_total=11000/s = 1.100e4 项访问/s
#         比值 = 3114×
