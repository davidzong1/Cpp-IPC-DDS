### §4.2-2 has_peers() 不再是 stale 扫描总门控（仅一处 if(has) 分支，含 else 兜底）
    if (has)
    {
        /* 有 peer：每拍扫，并把兜底的下次到期点推后（刚扫过，不必紧接着再扫）。 */
        st.next_stale_due = now + dead_timeout;
        pub.on_pub_stale_scan(now, dead_timeout);
        return;
    }

    /* 无 peer：低频兜底 —— 到期才扫。 */
    if (now < st.next_stale_due)
    {
        return;
    }
    st.next_stale_due = now + dead_timeout;
    pub.on_pub_stale_scan(now, dead_timeout);
}
